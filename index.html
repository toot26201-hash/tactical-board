function resizeCanvas() { const rect = container.getBoundingClientRect(); const padX = 24; const padY = 56; // مساحة كافية للشريط العلوي const availW = Math.max(300, rect.width - padX * 2); const availH = Math.max(250, rect.height - padY); // النسبة القانونية الدقيقة للملعب (105 متر طول × 68 متر عرض) const targetRatio = pitchMode === 'full' ? (105 / 68) : (68 / 52.5); // حساب الأبعاد بناءً على الارتفاع أولاً لضمان عدم تمدد الملعب أفقياً let finalH = availH; let finalW = finalH * targetRatio; if (finalW > availW) { finalW = availW; finalH = finalW / targetRatio; } const dpr = window.devicePixelRatio || 1; canvas.width = Math.round(finalW * dpr); canvas.height = Math.round(finalH * dpr); canvas.style.width = Math.round(finalW) + 'px'; canvas.style.height = Math.round(finalH) + 'px'; ctx.setTransform(dpr, 0, 0, dpr, 0, 0); const margin = Math.max(12, Math.min(finalW, finalH) * 0.025); pitchRect = { x: margin, y: margin, w: finalW - margin * 2, h: finalH - margin * 2, canvasW: finalW, canvasH: finalH }; render(); }
